// ========================================================================= // EKSEKUSI DATA PENDAFTARAN WAJAH KE GOOGLE SHEET // ========================================================================= btnSimpan.addEventListener('click', async () => { let nikFinal = document.getElementById('input-nik-terpilih').value; let namaFinal = document.getElementById('lbl-nama-terpilih').innerText; if (!nikFinal) { tampilkanPopUp("Pilih nama santri terlebih dahulu dari daftar!", "bg-amber-500 text-slate-950 border-amber-400"); return; } if (!posisiWajahTerakhir) { tampilkanPopUp("Wajah belum terdeteksi kamera! Dekatkan posisi wajah ke kotak tengah.", "bg-red-500 text-white border-red-400"); return; } btnSimpan.disabled = true; btnSimpan.innerText = "MENYIMPAN TITIK WAJAH KE GOOGLE SHEET..."; // Menggunakan URLSearchParams agar selaras dengan parameter pembacaan e.parameter biasa const params = new URLSearchParams(); params.append('aksi', 'daftar_wajah_saja'); params.append('nik', nikFinal); params.append('vektor', JSON.stringify(posisiWajahTerakhir)); // Dikirim dengan key 'vektor' sesuai sheet try { let res = await fetch(API_URL, { method: "POST", headers: { "Content-Type": "application/x-www-form-urlencoded" }, body: params.toString() }); let r = await res.json(); if (r.status === "success") { tampilkanPopUp(`✅ BERHASIL MENGAKTIFKAN!\nWajah asli ${namaFinal} telah dikunci ke Google Sheet.`, "bg-emerald-500 text-slate-950 border-emerald-400"); // Reset Input Form setelah sukses document.getElementById('input-nik-terpilih').value = ""; inputCari.value = ""; document.getElementById('info-santri-terpilih').classList.add('hidden'); } else { tampilkanPopUp(`❌ GAGAL: ${r.message}`, "bg-red-500 text-white border-red-400"); } } catch (err) { tampilkanPopUp("Gangguan koneksi internet! Gagal mengirim data ke cloud Google Sheets.", "bg-red-500 text-white border-red-400"); } finally { btnSimpan.disabled = false; btnSimpan.innerText = "KUNCI & SIMPAN WAJAH ASLI"; } });
